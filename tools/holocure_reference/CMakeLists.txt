cmake_minimum_required(VERSION 3.25)
project(HoloReference LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The reference recorder requires MSVC x64 (GameMaker's C++ ABI).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(YYTK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/YYToolkit/ExamplePlugin/include")
add_library(HoloReference SHARED recorder.cpp
  "${YYTK_INCLUDE}/YYToolkit/YYTK_Shared_Types.cpp")
target_include_directories(HoloReference PRIVATE "${YYTK_INCLUDE}")
target_compile_definitions(HoloReference PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(HoloReference PRIVATE /EHsc /W4)
add_library(HoloProfile SHARED profile_isolation.cpp)
target_compile_definitions(HoloProfile PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
target_link_libraries(HoloProfile PRIVATE shell32)
